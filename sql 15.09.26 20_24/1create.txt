-- 2.1. Таблица РОЛИ
CREATE TABLE "РОЛИ" (
    "id_роли" SERIAL PRIMARY KEY,
    "Название" VARCHAR(50) NOT NULL UNIQUE
);

-- 2.2. Таблица ПОЛЬЗОВАТЕЛИ
CREATE TABLE "ПОЛЬЗОВАТЕЛИ" (
    "id_пользователя" SERIAL PRIMARY KEY,
    "ник" VARCHAR(50) NOT NULL UNIQUE,
    "пароль" VARCHAR(255) NOT NULL,
    "id_роли" INT REFERENCES "РОЛИ"("id_роли") ON DELETE RESTRICT
);

-- 2.3. Таблица ЛОГИ
CREATE TABLE "ЛОГИ" (
    "id_лога" SERIAL PRIMARY KEY,
    "id_пользователя" INT REFERENCES "ПОЛЬЗОВАТЕЛИ"("id_пользователя") ON DELETE SET NULL,
    "Действие" VARCHAR(255) NOT NULL,
    "Дата_время" TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2.4. Таблица ПРЕПОДАВАТЕЛИ (со всеми полями со схемы)
CREATE TABLE "ПРЕПОДАВАТЕЛИ" (
    "id_преподавателя" SERIAL PRIMARY KEY,
    "id_пользователя" INT UNIQUE REFERENCES "ПОЛЬЗОВАТЕЛИ"("id_пользователя") ON DELETE SET NULL,
    "ФИО" VARCHAR(150) NOT NULL,
    "Дата_рождения" DATE,
    "Пол" VARCHAR(10),
    "Кафедра" VARCHAR(150),
    "Должность" VARCHAR(100),
    "Ученая_степень" VARCHAR(100),
    "Контакты" VARCHAR(255)
);

-- 2.5. Таблица ГРУППЫ
CREATE TABLE "ГРУППЫ" (
    "id_группы" SERIAL PRIMARY KEY,
    "id_Куратора" INT REFERENCES "ПРЕПОДАВАТЕЛИ"("id_преподавателя") ON DELETE SET NULL,
    "Название" VARCHAR(50) NOT NULL UNIQUE,
    "Специальность" VARCHAR(150),
    "Курс" INT CHECK ("Курс" BETWEEN 1 AND 6),
    "Форма_обучения" VARCHAR(50),
    "Год_набора" INT
);

-- 2.6. Таблица СТУДЕНТЫ
CREATE TABLE "СТУДЕНТЫ" (
    "id_студента" SERIAL PRIMARY KEY,
    "id_группы" INT REFERENCES "ГРУППЫ"("id_группы") ON DELETE CASCADE,
    "id_пользователя" INT UNIQUE REFERENCES "ПОЛЬЗОВАТЕЛИ"("id_пользователя") ON DELETE SET NULL,
    "ФИО" VARCHAR(150) NOT NULL,
    "Дата_рождения" DATE,
    "Пол" VARCHAR(10),
    "Контакты" VARCHAR(255),
    "Форма_обучения" VARCHAR(50),
    "Дата_поступления" DATE,
    "Статус" VARCHAR(50) DEFAULT 'Обучается'
);

-- 2.7. Таблица ПРЕДМЕТЫ
CREATE TABLE "ПРЕДМЕТЫ" (
    "id_предмета" SERIAL PRIMARY KEY,
    "Название" VARCHAR(150) NOT NULL,
    "описание" TEXT,
    "Часы_лекций" INT DEFAULT 0,
    "Часы_практик" INT DEFAULT 0,
    "Часы_лабораторных" INT DEFAULT 0,
    "Форма_контроля" VARCHAR(50)
);

-- 2.8. Таблица ПОТОК
CREATE TABLE "ПОТОК" (
    "id_потока" SERIAL PRIMARY KEY,
    "id_группы" INT REFERENCES "ГРУППЫ"("id_группы") ON DELETE CASCADE,
    "id_предмета" INT REFERENCES "ПРЕДМЕТЫ"("id_предмета") ON DELETE CASCADE,
    "id_преподавателя" INT REFERENCES "ПРЕПОДАВАТЕЛИ"("id_преподавателя") ON DELETE SET NULL,
    "Семестр" INT,
    "Учебный_год" VARCHAR(20)
);

-- 2.9. Таблица ОЦЕНКИ
CREATE TABLE "ОЦЕНКИ" (
    "id_оценки" SERIAL PRIMARY KEY,
    "id_студента" INT REFERENCES "СТУДЕНТЫ"("id_студента") ON DELETE CASCADE,
    "id_потока" INT REFERENCES "ПОТОК"("id_потока") ON DELETE CASCADE,
    "Тип_оценки" VARCHAR(50),
    "Оценка" VARCHAR(10),
    "Вес_оценки" NUMERIC(3,2) DEFAULT 1.00,
    "Дата_выставления" DATE DEFAULT CURRENT_DATE
);

-- 2.10. Таблица ПОСЕЩАЕМОСТЬ
CREATE TABLE "ПОСЕЩАЕМОСТЬ" (
    "id_посещаемости" SERIAL PRIMARY KEY,
    "id_студента" INT REFERENCES "СТУДЕНТЫ"("id_студента") ON DELETE CASCADE,
    "id_дисциплины_группы" INT REFERENCES "ПОТОК"("id_потока") ON DELETE CASCADE,
    "дата_занятия" DATE DEFAULT CURRENT_DATE,
    "статус" VARCHAR(20)
);
